        private const string TableText = "@TEXT@";

        private static Dictionary<string, Cell[]> table;

        private static Dictionary<string, Cell[]> Table
        {
            get
            {
                if(table == null)
                {
                    var t = new Dictionary<string, Cell[]>();
                    foreach(var line in TableText.Split('\n'))
                    {
                        var f = line.Split('\t');
                        t[f[0]] = f.Skip(1).Select(x => { var at = x.LastIndexOf('@'); return new Cell(x.Substring(0, at), int.Parse(x.Substring(at + 1))); }).ToArray();
                    }
                    table = t;
                }
                return table;
            }
        }
